Secure Storage

A10 — Lesson 1 of 2  |  ~35 min

Why this matters for Token: AsyncStorage writes data as plaintext files on the device filesystem. If someone gets physical access to a rooted/jailbroken phone (or even a locked phone with USB debugging enabled), they can read the JWT, refresh token, and any encryption keys you stored there. Token is a privacy app — its credibility depends on actually protecting secrets. expo-secure-store uses iOS Keychain and Android Keystore hardware-backed encryption so that even a rooted device cannot trivially extract these values.

AsyncStorage vs SecureStore

AsyncStorage SecureStore
Encryption None (plaintext JSON file) Hardware-backed (Keychain/Keystore)
Size limit 6 MB (Android default) 2048 bytes per item (iOS)
Use for Preferences, cache, non-sensitive data JWT, refresh token, encryption keys
Availability Always available May fail after device wipe or restore
Speed Fast (file I/O) Slightly slower (crypto operations)

The rule is simple: if the value is a secret, use SecureStore. If it is a preference or cached UI state, use AsyncStorage.

Installing expo-secure-store

# In your app/ directory
npx expo install expo-secure-store

No additional native configuration needed with Expo managed workflow. The package uses iOS Keychain Services and Android EncryptedSharedPreferences (API 23+) or Keystore (API 18+).

Basic SecureStore API

import * as SecureStore from 'expo-secure-store';

// Store a value (must be a string, max 2048 bytes on iOS)
await SecureStore.setItemAsync('auth_token', jwtToken);

// Retrieve a value (returns null if not found)
const token = await SecureStore.getItemAsync('auth_token');

// Delete a value
await SecureStore.deleteItemAsync('auth_token');

// Options: control when the item is accessible
await SecureStore.setItemAsync('refresh_token', refreshToken, {
  // WHEN_UNLOCKED — accessible only when device is unlocked (default)
  // AFTER_FIRST_UNLOCK — accessible after first unlock since boot
  // ALWAYS — accessible even when locked (least secure)
  keychainAccessible: SecureStore.WHEN_UNLOCKED,
});

Building a Secure Storage Wrapper

// app/src/services/secureStorage.ts
import * as SecureStore from 'expo-secure-store';

export class SecureStorageError extends Error {
  constructor(message: string, public readonly key: string) {
    super(message);
    this.name = 'SecureStorageError';
  }
}

/**
 * Wrapper around expo-secure-store with error handling
 * and JSON serialization for objects.
 */
export const secureStorage = {
  async set(key: string, value: string): Promise<void> {
    try {
      if (value.length > 2048) {
        throw new SecureStorageError(
          `Value exceeds 2048 byte iOS limit (${value.length} bytes)`,
          key
        );
      }
      await SecureStore.setItemAsync(key, value);
    } catch (err) {
      if (err instanceof SecureStorageError) throw err;
      throw new SecureStorageError(
        `Failed to store key "${key}": ${(err as Error).message}`,
        key
      );
    }
  },

  async get(key: string): Promise<string | null> {
    try {
      return await SecureStore.getItemAsync(key);
    } catch (err) {
      // Keychain may be unavailable after device restore
      console.error(`SecureStore read failed for "${key}":`, err);
      return null;
    }
  },

  async remove(key: string): Promise<void> {
    try {
      await SecureStore.deleteItemAsync(key);
    } catch (err) {
      console.error(`SecureStore delete failed for "${key}":`, err);
    }
  },

  async setJSON(key: string, value: unknown): Promise<void> {
    const serialized = JSON.stringify(value);
    await this.set(key, serialized);
  },

  async getJSON<T>(key: string): Promise<T | null> {
    const raw = await this.get(key);
    if (!raw) return null;
    try {
      return JSON.parse(raw) as T;
    } catch {
      return null;
    }
  },
};

Migrating from AsyncStorage to SecureStore

// app/src/services/migrateSensitiveData.ts
import AsyncStorage from '@react-native-async-storage/async-storage';
import { secureStorage } from './secureStorage';

const MIGRATION_FLAG = '@token/secure_migration_v1';

/**
 * One-time migration: moves sensitive keys from AsyncStorage
 * to SecureStore. Safe to call multiple times — checks a flag.
 */
export async function migrateSensitiveData(): Promise<void> {
  const alreadyMigrated = await AsyncStorage.getItem(MIGRATION_FLAG);
  if (alreadyMigrated === 'done') return;

  const keysToMigrate = [
    { asyncKey: '@token/jwt', secureKey: 'token_auth_jwt' },
    { asyncKey: '@token/refresh', secureKey: 'token_refresh_jwt' },
    { asyncKey: '@token/e2e_key', secureKey: 'token_e2e_key' },
  ];

  for (const { asyncKey, secureKey } of keysToMigrate) {
    const value = await AsyncStorage.getItem(asyncKey);
    if (value) {
      await secureStorage.set(secureKey, value);
      await AsyncStorage.removeItem(asyncKey); // Delete plaintext copy
    }
  }

  await AsyncStorage.setItem(MIGRATION_FLAG, 'done');
  console.log('Sensitive data migration complete');
}

Using SecureStore in Your Auth Context

// app/src/context/AuthContext.tsx (updated)
import { secureStorage } from '../services/secureStorage';

const KEYS = {
  AUTH: 'token_auth_jwt',
  REFRESH: 'token_refresh_jwt',
};

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  // On app start: try to restore session from SecureStore
  useEffect(() => {
    async function restoreSession() {
      const jwt = await secureStorage.get(KEYS.AUTH);
      if (jwt) {
        try {
          const userData = await apiClient.get('/api/me', jwt);
          setUser(userData);
        } catch {
          // Token expired — try refresh
          const refreshToken = await secureStorage.get(KEYS.REFRESH);
          if (refreshToken) {
            const newTokens = await apiClient.post('/api/refresh', { refreshToken });
            await secureStorage.set(KEYS.AUTH, newTokens.accessToken);
            await secureStorage.set(KEYS.REFRESH, newTokens.refreshToken);
            setUser(newTokens.user);
          }
        }
      }
      setLoading(false);
    }
    restoreSession();
  }, []);

  async function login(email: string, password: string) {
    const response = await apiClient.post('/api/login', { email, password });
    await secureStorage.set(KEYS.AUTH, response.accessToken);
    await secureStorage.set(KEYS.REFRESH, response.refreshToken);
    setUser(response.user);
  }

  async function logout() {
    await secureStorage.remove(KEYS.AUTH);
    await secureStorage.remove(KEYS.REFRESH);
    setUser(null);
  }

  return (
    <AuthContext.Provider value={{ user, loading, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

Error Handling: When the Keychain is Unavailable

// After a device restore or factory reset, the Keychain may reject reads.
// Your app must handle this gracefully — force re-login.

async function safeGetToken(): Promise<string | null> {
  try {
    return await SecureStore.getItemAsync('token_auth_jwt');
  } catch (err) {
    // Keychain corrupted or inaccessible
    console.error('Keychain unavailable:', err);
    // Clear all secure data and force re-authentication
    await clearAllSecureData();
    return null;
  }
}

async function clearAllSecureData(): Promise<void> {
  const keys = ['token_auth_jwt', 'token_refresh_jwt', 'token_e2e_key'];
  for (const key of keys) {
    try {
      await SecureStore.deleteItemAsync(key);
    } catch {
      // Already inaccessible — that's fine
    }
  }
}
