Biometric App Lock

A10 — Lesson 2 of 2  |  ~40 min

Why this matters for Token: Token is a privacy-first app. If someone picks up your unlocked phone, they should not be able to see your token list, read messages, or impersonate you. A biometric lock screen (fingerprint or Face ID) adds a second barrier — the phone being unlocked is not enough, the app itself requires your biometric identity. This is especially important because Token holds the power to create, revoke, and manage all access to the user.

How Biometric Authentication Works on Mobile

Both iOS and Android provide system-level biometric APIs:

  • iOS: Face ID or Touch ID via the LocalAuthentication framework
  • Android: Fingerprint, face, or iris via BiometricPrompt API

expo-local-authentication wraps both behind one API. The biometric data never leaves the device — you are not storing fingerprints. You are asking the OS: "Is this the device owner?" and getting back a yes/no answer.

Checking Biometric Availability

import * as LocalAuthentication from 'expo-local-authentication';

// Step 1: Does the device have biometric hardware?
const hasHardware = await LocalAuthentication.hasHardwareAsync();
// true on most modern phones, false on old devices or emulators

// Step 2: Is any biometric enrolled (registered)?
const isEnrolled = await LocalAuthentication.isEnrolledAsync();
// false if user hasn't set up fingerprint/Face ID in system settings

// Step 3: What types are available?
const types = await LocalAuthentication.supportedAuthenticationTypesAsync();
// [1] = fingerprint, [2] = facial recognition, [3] = iris

// Combine into a single check
export async function canUseBiometrics(): Promise<boolean> {
  const hardware = await LocalAuthentication.hasHardwareAsync();
  const enrolled = await LocalAuthentication.isEnrolledAsync();
  return hardware && enrolled;
}

Performing Authentication

export async function authenticate(reason?: string): Promise<boolean> {
  const result = await LocalAuthentication.authenticateAsync({
    promptMessage: reason || 'Verify your identity to open Token',
    // iOS: allow fallback to device passcode
    fallbackLabel: 'Use Passcode',
    // Android: allow device PIN/pattern/password as fallback
    disableDeviceCredentialFallback: false,
    // Cancel button label
    cancelLabel: 'Cancel',
  });

  if (result.success) {
    return true;
  }

  // result.error can be:
  // 'user_cancel' — user pressed cancel
  // 'system_cancel' — OS interrupted (incoming call)
  // 'user_fallback' — user chose passcode fallback
  // 'lockout' — too many failed attempts
  // 'not_enrolled' — no biometric registered
  // 'not_available' — hardware not available

  if (result.error === 'lockout') {
    // Device is temporarily locked — inform the user
    console.warn('Too many failed attempts. Try again later.');
  }

  return false;
}

The Lock Screen Component

// app/src/components/LockScreen.tsx
import React, { useState, useEffect } from 'react';
import { View, Text, StyleSheet, TouchableOpacity } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { authenticate } from '../services/biometricAuth';

interface LockScreenProps {
  onUnlock: () => void;
}

export function LockScreen({ onUnlock }: LockScreenProps) {
  const [error, setError] = useState<string | null>(null);
  const [attempting, setAttempting] = useState(false);

  // Attempt authentication on mount
  useEffect(() => {
    attemptUnlock();
  }, []);

  async function attemptUnlock() {
    setAttempting(true);
    setError(null);

    const success = await authenticate('Unlock Token');
    if (success) {
      onUnlock();
    } else {
      setError('Authentication failed. Tap to try again.');
    }
    setAttempting(false);
  }

  return (
    <View style={styles.container}>
      <Ionicons name="lock-closed" size={64} color="#4F46E5" />
      <Text style={styles.title}>Token is Locked</Text>
      <Text style={styles.subtitle}>
        Verify your identity to continue
      </Text>

      {error && (
        <TouchableOpacity onPress={attemptUnlock} disabled={attempting}>
          <Text style={styles.error}>{error}</Text>
          <Text style={styles.retry}>Tap to retry</Text>
        </TouchableOpacity>
      )}
    </View>
  );
}

Integrating the Lock Screen with App State

// app/src/providers/AppLockProvider.tsx
import React, { useState, useEffect, useRef, useCallback } from 'react';
import { AppState, AppStateStatus } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { LockScreen } from '../components/LockScreen';
import { canUseBiometrics } from '../services/biometricAuth';

const LOCK_TIMEOUT_KEY = '@token/lock_timeout'; // seconds
const LOCK_ENABLED_KEY = '@token/lock_enabled';
const DEFAULT_TIMEOUT = 0; // 0 = lock immediately on background

interface AppLockProviderProps {
  children: React.ReactNode;
}

export function AppLockProvider({ children }: AppLockProviderProps) {
  const [locked, setLocked] = useState(true);
  const [lockEnabled, setLockEnabled] = useState(false);
  const [loading, setLoading] = useState(true);
  const backgroundTimestamp = useRef<number | null>(null);

  useEffect(() => {
    checkLockSetting();
  }, []);

  async function checkLockSetting() {
    const enabled = await AsyncStorage.getItem(LOCK_ENABLED_KEY);
    const biometricsAvailable = await canUseBiometrics();

    if (enabled === 'true' && biometricsAvailable) {
      setLockEnabled(true);
      setLocked(true); // Start locked
    } else {
      setLockEnabled(false);
      setLocked(false); // No lock configured
    }
    setLoading(false);
  }

  // Track app state changes
  useEffect(() => {
    if (!lockEnabled) return;

    const subscription = AppState.addEventListener('change', handleAppState);
    return () => subscription.remove();
  }, [lockEnabled]);

  const handleAppState = useCallback(async (state: AppStateStatus) => {
    if (state === 'background' || state === 'inactive') {
      backgroundTimestamp.current = Date.now();
    } else if (state === 'active' && backgroundTimestamp.current) {
      const elapsed = (Date.now() - backgroundTimestamp.current) / 1000;
      const timeoutStr = await AsyncStorage.getItem(LOCK_TIMEOUT_KEY);
      const timeout = timeoutStr ? parseInt(timeoutStr, 10) : DEFAULT_TIMEOUT;

      if (elapsed >= timeout) {
        setLocked(true);
      }
      backgroundTimestamp.current = null;
    }
  }, []);

  function handleUnlock() {
    setLocked(false);
  }

  if (loading) return null; // Or a splash screen

  if (locked && lockEnabled) {
    return <LockScreen onUnlock={handleUnlock} />;
  }

  return <>{children}</>;
}

Settings: Enable/Disable and Timeout

// app/src/screens/SecuritySettings.tsx
import React, { useState, useEffect } from 'react';
import { View, Text, Switch, StyleSheet } from 'react-native';
import { Picker } from '@react-native-picker/picker';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { canUseBiometrics } from '../services/biometricAuth';
import { authenticate } from '../services/biometricAuth';

export function SecuritySettings() {
  const [biometricAvailable, setBiometricAvailable] = useState(false);
  const [lockEnabled, setLockEnabled] = useState(false);
  const [timeout, setTimeout] = useState('0');

  useEffect(() => {
    loadSettings();
  }, []);

  async function loadSettings() {
    const available = await canUseBiometrics();
    setBiometricAvailable(available);

    const enabled = await AsyncStorage.getItem('@token/lock_enabled');
    setLockEnabled(enabled === 'true');

    const savedTimeout = await AsyncStorage.getItem('@token/lock_timeout');
    setTimeout(savedTimeout || '0');
  }

  async function toggleLock(value: boolean) {
    if (value) {
      // Require biometric verification to ENABLE the lock
      const verified = await authenticate('Verify identity to enable app lock');
      if (!verified) return;
    }

    await AsyncStorage.setItem('@token/lock_enabled', value ? 'true' : 'false');
    setLockEnabled(value);
  }

  async function changeTimeout(value: string) {
    await AsyncStorage.setItem('@token/lock_timeout', value);
    setTimeout(value);
  }

  return (
    <View style={styles.container}>
      <Text style={styles.heading}>App Lock</Text>

      {!biometricAvailable && (
        <Text style={styles.warning}>
          No biometric authentication available. Set up fingerprint or
          Face ID in your device settings first.
        </Text>
      )}

      <View style={styles.row}>
        <Text style={styles.label}>Require biometric to open</Text>
        <Switch
          value={lockEnabled}
          onValueChange={toggleLock}
          disabled={!biometricAvailable}
        />
      </View>

      {lockEnabled && (
        <View style={styles.row}>
          <Text style={styles.label}>Lock after</Text>
          <Picker
            selectedValue={timeout}
            onValueChange={changeTimeout}
            style={styles.picker}
          >
            <Picker.Item label="Immediately" value="0" />
            <Picker.Item label="30 seconds" value="30" />
            <Picker.Item label="1 minute" value="60" />
            <Picker.Item label="5 minutes" value="300" />
          </Picker>
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 16, backgroundColor: '#0F0F0F' },
  heading: { fontSize: 20, fontWeight: '700', color: '#FFF', marginBottom: 16 },
  warning: { color: '#F59E0B', marginBottom: 16, fontSize: 14 },
  row: {
    flexDirection: 'row', justifyContent: 'space-between',
    alignItems: 'center', paddingVertical: 12,
    borderBottomWidth: 1, borderBottomColor: '#1F1F1F',
  },
  label: { fontSize: 16, color: '#E5E7EB' },
  picker: { width: 160, color: '#E5E7EB' },
});

Handling Biometric Changes

// If a new fingerprint is added, should the app re-verify?
// iOS: adding a new fingerprint invalidates Keychain items stored with
//       kSecAccessControlBiometryCurrentSet — but SecureStore uses a
//       less restrictive policy by default.
// Android: new fingerprints don't invalidate existing keys by default.

// Strategy for Token: when the user opens the app after a biometric change,
// require the full device passcode once, then continue normally.

export async function checkBiometricIntegrity(): Promise<'ok' | 'changed' | 'unavailable'> {
  const available = await canUseBiometrics();
  if (!available) return 'unavailable';

  // We store a sentinel in SecureStore with biometryCurrentSet policy
  // If reading it fails, biometrics changed since we wrote it
  try {
    const sentinel = await SecureStore.getItemAsync('bio_sentinel', {
      keychainAccessible: SecureStore.WHEN_UNLOCKED,
    });
    if (sentinel === null) {
      // First time or was cleared — set it
      await SecureStore.setItemAsync('bio_sentinel', 'valid');
      return 'ok';
    }
    return 'ok';
  } catch {
    return 'changed';
  }
}

UX Consideration: Don't Lock During Active Calls

// app/src/providers/AppLockProvider.tsx — enhanced handleAppState

import { useCallState } from '../context/CallContext';

// Inside the provider:
const { activeCall } = useCallState();

const handleAppState = useCallback(async (state: AppStateStatus) => {
  // Never lock during an active call — the user may switch apps
  // to check something and come back
  if (activeCall) {
    backgroundTimestamp.current = null;
    return;
  }

  if (state === 'background' || state === 'inactive') {
    backgroundTimestamp.current = Date.now();
  } else if (state === 'active' && backgroundTimestamp.current) {
    const elapsed = (Date.now() - backgroundTimestamp.current) / 1000;
    const timeoutStr = await AsyncStorage.getItem(LOCK_TIMEOUT_KEY);
    const timeout = timeoutStr ? parseInt(timeoutStr, 10) : DEFAULT_TIMEOUT;

    if (elapsed >= timeout) {
      setLocked(true);
    }
    backgroundTimestamp.current = null;
  }
}, [activeCall]);

Hiding Content in App Switcher

// When the app goes to background, iOS and Android take a screenshot
// for the app switcher. This can expose sensitive content.

import { AppState, View, StyleSheet } from 'react-native';
import { useState, useEffect } from 'react';

export function PrivacyOverlay({ children }: { children: React.ReactNode }) {
  const [showOverlay, setShowOverlay] = useState(false);

  useEffect(() => {
    const subscription = AppState.addEventListener('change', (state) => {
      // Show a blank overlay when going to background
      setShowOverlay(state === 'inactive' || state === 'background');
    });
    return () => subscription.remove();
  }, []);

  return (
    <View style={{ flex: 1 }}>
      {children}
      {showOverlay && <View style={styles.overlay} />}
    </View>
  );
}

const styles = StyleSheet.create({
  overlay: {
    ...StyleSheet.absoluteFillObject,
    backgroundColor: '#0F0F0F', // Matches app background
    zIndex: 9999,
  },
});
