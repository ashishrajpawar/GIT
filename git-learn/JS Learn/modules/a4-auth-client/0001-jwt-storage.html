JWT Flow — Login, Signup, Token Storage

A4 — Lesson 1 of 3 ~50 min
Why this matters for Token

The mobile app needs to log in, store the JWT tokens securely, and send the access token with every API request. Where you store these tokens determines your app's security posture. AsyncStorage is unencrypted plaintext — anyone with device access can read it. expo-secure-store uses the device's hardware-backed keychain (iOS) or keystore (Android). This lesson builds the client-side half of the auth system from B4.

The client-side auth flow

  1. User opens app → check if refresh token exists in secure storage
  2. If no token → show Login/Register screens
  3. User submits credentials → API returns access + refresh tokens
  4. Store refresh token in secure storage, access token in memory
  5. All subsequent API calls include the access token in Authorization header
  6. When access token expires (15 min) → silently refresh using the refresh token

expo-secure-store — hardware-backed encryption

npx expo install expo-secure-store
import * as SecureStore from 'expo-secure-store';

// Store a value (encrypted by iOS Keychain / Android Keystore)
await SecureStore.setItemAsync('refreshToken', token);

// Read a value
const token = await SecureStore.getItemAsync('refreshToken');

// Delete a value
await SecureStore.deleteItemAsync('refreshToken');
Never use AsyncStorage for secrets

AsyncStorage stores data as unencrypted JSON files on disk. A rooted device, a backup extraction, or a malicious app with storage access can read everything. Refresh tokens, API keys, and any credential must go in SecureStore. AsyncStorage is fine for non-sensitive data: theme preference, onboarding completion, cached display names.

Token storage strategy

What Where Why
Refresh token SecureStore Long-lived, must survive app restarts, security-critical
Access token In-memory variable Short-lived (15 min), no persistence needed, gone on app close
User profile In-memory (React state) Refetched on every app launch, not sensitive

Building the auth store

// app/src/auth/store.ts
import * as SecureStore from 'expo-secure-store';

const REFRESH_KEY = 'token_refresh';
let accessToken: string | null = null;

export function getAccessToken(): string | null {
  return accessToken;
}

export function setAccessToken(token: string | null) {
  accessToken = token;
}

export async function getRefreshToken(): Promise<string | null> {
  return SecureStore.getItemAsync(REFRESH_KEY);
}

export async function setRefreshToken(token: string): Promise<void> {
  await SecureStore.setItemAsync(REFRESH_KEY, token);
}

export async function clearTokens(): Promise<void> {
  accessToken = null;
  await SecureStore.deleteItemAsync(REFRESH_KEY);
}

Login screen

// app/src/screens/LoginScreen.tsx
import { useState } from 'react';
import { View, Text, TextInput, Pressable, Alert } from 'react-native';
import { setAccessToken, setRefreshToken } from '../auth/store';

export function LoginScreen({ navigation }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);

  async function handleLogin() {
    setLoading(true);
    try {
      const res = await fetch('http://localhost:3000/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
      });

      const json = await res.json();

      if (!res.ok) {
        Alert.alert('Login failed', json.error);
        return;
      }

      // Store tokens
      setAccessToken(json.data.accessToken);
      await setRefreshToken(json.data.refreshToken);

      // Navigate to main app
      navigation.replace('Main');
    } catch (err) {
      Alert.alert('Network error', 'Could not connect to server');
    } finally {
      setLoading(false);
    }
  }

  return (
    <View style={{ flex: 1, padding: 24, justifyContent: 'center' }}>
      <Text style={{ fontSize: 28, fontWeight: '700', marginBottom: 32 }}>
        Sign in to Token
      </Text>

      <TextInput
        placeholder="Email"
        value={email}
        onChangeText={setEmail}
        autoCapitalize="none"
        keyboardType="email-address"
        style={inputStyle}
      />

      <TextInput
        placeholder="Password"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
        style={inputStyle}
      />

      <Pressable
        onPress={handleLogin}
        disabled={loading}
        style={[buttonStyle, loading && { opacity: 0.6 }]}
      >
        <Text style={{ color: '#fff', fontWeight: '600', fontSize: 16 }}>
          {loading ? 'Signing in...' : 'Sign In'}
        </Text>
      </Pressable>

      <Pressable onPress={() => navigation.navigate('Register')}>
        <Text style={{ textAlign: 'center', marginTop: 16, color: '#2563EB' }}>
          Don't have an account? Register
        </Text>
      </Pressable>
    </View>
  );
}

const inputStyle = {
  borderWidth: 1, borderColor: '#E5E7EB', borderRadius: 8,
  padding: 14, fontSize: 16, marginBottom: 12,
};
const buttonStyle = {
  backgroundColor: '#2563EB', padding: 16, borderRadius: 8,
  alignItems: 'center', marginTop: 8,
};

Registration screen

// app/src/screens/RegisterScreen.tsx
export function RegisterScreen({ navigation }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [displayName, setDisplayName] = useState('');
  const [loading, setLoading] = useState(false);

  async function handleRegister() {
    setLoading(true);
    try {
      const res = await fetch('http://localhost:3000/api/auth/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password, displayName }),
      });

      const json = await res.json();

      if (!res.ok) {
        Alert.alert('Registration failed', json.error || json.details?.[0]?.message);
        return;
      }

      // Auto-login after successful registration
      Alert.alert('Success', 'Account created! Please sign in.');
      navigation.replace('Login');
    } catch (err) {
      Alert.alert('Network error', 'Could not connect to server');
    } finally {
      setLoading(false);
    }
  }

  // ... similar UI to LoginScreen with displayName field added
}

App startup — checking existing session

// app/src/screens/SplashScreen.tsx
import { useEffect } from 'react';
import { getRefreshToken, setAccessToken, setRefreshToken } from '../auth/store';

export function SplashScreen({ navigation }) {
  useEffect(() => {
    checkSession();
  }, []);

  async function checkSession() {
    const refreshToken = await getRefreshToken();

    if (!refreshToken) {
      // No saved session — go to login
      navigation.replace('Login');
      return;
    }

    // Try to get a new access token using the refresh token
    try {
      const res = await fetch('http://localhost:3000/api/auth/refresh', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refreshToken }),
      });

      if (!res.ok) {
        // Refresh token expired or revoked — go to login
        navigation.replace('Login');
        return;
      }

      const json = await res.json();
      setAccessToken(json.data.accessToken);
      await setRefreshToken(json.data.refreshToken);

      navigation.replace('Main');
    } catch {
      // Network error — could show offline screen or retry
      navigation.replace('Login');
    }
  }

  return null; // Or a splash/loading screen
}
