Auth Context & Protected Routes

A4 — Lesson 2 of 3 ~50 min
Why this matters for Token

Every screen in Token needs to know: is the user logged in? What's their name? What's their ID? Passing this through props would mean threading auth data through 10 layers of components. React Context solves this — define auth state once at the top, access it from any component with a hook. And the navigator automatically shows Login vs Main screens based on auth state.

React Context — the concept

Context provides values to an entire component tree without prop drilling:

// 1. Create the context
const AuthContext = createContext(defaultValue);

// 2. Wrap your app in a Provider (supplies the value)
<AuthContext.Provider value={authState}>
  <App />
</AuthContext.Provider>

// 3. Any nested component reads it with useContext
const auth = useContext(AuthContext);

Token's AuthContext

// app/src/auth/AuthContext.tsx
import { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import * as SecureStore from 'expo-secure-store';
import { setAccessToken, clearTokens } from './store';

interface User {
  id: number;
  email: string;
  displayName: string;
}

interface AuthState {
  user: User | null;
  isLoading: boolean;
  login: (email: string, password: string) => Promise<void>;
  register: (email: string, password: string, displayName: string) => Promise<void>;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthState | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  // Check for existing session on app launch
  useEffect(() => {
    restoreSession();
  }, []);

  async function restoreSession() {
    try {
      const refreshToken = await SecureStore.getItemAsync('refreshToken');
      if (!refreshToken) {
        setIsLoading(false);
        return;
      }

      const res = await fetch('http://localhost:3000/api/auth/refresh', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refreshToken }),
      });

      if (!res.ok) {
        await clearTokens();
        setIsLoading(false);
        return;
      }

      const json = await res.json();
      setAccessToken(json.data.accessToken);
      await SecureStore.setItemAsync('refreshToken', json.data.refreshToken);
      setUser(json.data.user);
    } catch {
      // Network error — stay on loading or show offline
    } finally {
      setIsLoading(false);
    }
  }

  async function login(email: string, password: string) {
    const res = await fetch('http://localhost:3000/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password }),
    });

    const json = await res.json();
    if (!res.ok) throw new Error(json.error);

    setAccessToken(json.data.accessToken);
    await SecureStore.setItemAsync('refreshToken', json.data.refreshToken);
    setUser(json.data.user);
  }

  async function register(email: string, password: string, displayName: string) {
    const res = await fetch('http://localhost:3000/api/auth/register', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password, displayName }),
    });

    const json = await res.json();
    if (!res.ok) throw new Error(json.error || json.details?.[0]?.message);

    // Auto-login after registration
    await login(email, password);
  }

  async function logout() {
    try {
      const token = await SecureStore.getItemAsync('refreshToken');
      // Tell server to revoke (best effort)
      await fetch('http://localhost:3000/api/auth/logout', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${token}` },
      });
    } catch {} // Don't block logout on network error

    await clearTokens();
    setUser(null);
  }

  return (
    <AuthContext.Provider value={{ user, isLoading, login, register, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

// Custom hook for easy access
export function useAuth(): AuthState {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth must be used within AuthProvider');
  return context;
}

Conditional navigation based on auth state

// app/src/navigation/RootNavigator.tsx
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { useAuth } from '../auth/AuthContext';
import { LoginScreen } from '../screens/LoginScreen';
import { RegisterScreen } from '../screens/RegisterScreen';
import { MainTabs } from './MainTabs';
import { LoadingSpinner } from '../components/LoadingSpinner';

const Stack = createNativeStackNavigator();

export function RootNavigator() {
  const { user, isLoading } = useAuth();

  if (isLoading) {
    return <LoadingSpinner message="Starting Token..." />;
  }

  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        {user ? (
          // Authenticated — show main app
          <Stack.Screen name="Main" component={MainTabs} />
        ) : (
          // Not authenticated — show auth screens
          <>
            <Stack.Screen name="Login" component={LoginScreen} />
            <Stack.Screen name="Register" component={RegisterScreen} />
          </>
        )}
      </Stack.Navigator>
    </NavigationContainer>
  );
}

When user changes from null to a User object (login success), React Navigation automatically swaps the screen stack. No manual navigation.replace() needed — the conditional rendering handles it.

App entry point

// app/App.tsx
import { AuthProvider } from './src/auth/AuthContext';
import { RootNavigator } from './src/navigation/RootNavigator';

export default function App() {
  return (
    <AuthProvider>
      <RootNavigator />
    </AuthProvider>
  );
}

Using auth in any component

// Any screen — no prop drilling needed
import { useAuth } from '../auth/AuthContext';

function SettingsScreen() {
  const { user, logout } = useAuth();

  return (
    <View>
      <Text>Welcome, {user?.displayName}</Text>
      <Text>Email: {user?.email}</Text>
      <Pressable onPress={logout}>
        <Text>Log Out</Text>
      </Pressable>
    </View>
  );
}

// In the API client — access token from the store
function TokenDetailScreen() {
  const { user } = useAuth();
  // user.id is available for any user-specific logic
}

Updated Login screen using AuthContext

// Simpler — all auth logic lives in the context
function LoginScreen() {
  const { login } = useAuth();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);

  async function handleSubmit() {
    setLoading(true);
    setError('');
    try {
      await login(email, password);
      // No navigation needed — RootNavigator reacts to user state change
    } catch (err: any) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }

  // ... render form with error display
}
