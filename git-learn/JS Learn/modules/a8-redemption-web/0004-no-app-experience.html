No-App-Installed Experience

A8 — Lesson 4 of 4 ~50 min
Why this matters for Token

The whole point of the redemption web page: holders communicate without installing anything. The web page IS the product for them. A delivery driver opens a link, types their name, and chats with the customer — from any browser, any device, any OS. If the mobile app happens to be installed, deep link into it instead. But the web experience must be complete and polished on its own.

The holder's complete journey

From the holder's perspective, the entire experience is:

  1. Receive a token code (QR, typed, spoken, printed)
  2. Open tokn.app/t/CODE in their phone's browser
  3. Enter their name (2 seconds)
  4. Chat or call — no sign-up, no download, no permissions upfront

This must work on Chrome, Safari, Firefox — desktop and mobile. Every extra step loses holders.

Chat UI in the browser

The ChatPage component is similar to what you built in the mobile app (A6), but using HTML/CSS instead of React Native Views:

// web/src/pages/ChatPage.tsx
import { useState, useEffect, useRef } from 'react';
import { useParams } from 'react-router-dom';
import { useWebSocket } from '../hooks/useWebSocket';

interface Message {
  id: string;
  text: string;
  sender: 'holder' | 'issuer';
  timestamp: string;
}

export function ChatPage() {
  const { code } = useParams<{ code: string }>();
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState('');
  const messagesEndRef = useRef<HTMLDivElement>(null);

  const token = sessionStorage.getItem('holderJwt');
  const wsClient = useWebSocket(import.meta.env.VITE_WS_URL, token);

  // Listen for incoming messages
  useEffect(() => {
    if (!wsClient.current) return;

    wsClient.current.onMessage((msg: any) => {
      if (msg.type === 'chat-message') {
        setMessages(prev => [...prev, {
          id: msg.id,
          text: msg.text,
          sender: msg.sender,
          timestamp: msg.timestamp,
        }]);
      }
    });
  }, [wsClient]);

  // Auto-scroll to bottom on new messages
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  function sendMessage(e: React.FormEvent) {
    e.preventDefault();
    if (!input.trim() || !wsClient.current) return;

    wsClient.current.send('chat-message', {
      text: input.trim(),
      tokenCode: code,
    });

    // Optimistic: add to local state immediately
    setMessages(prev => [...prev, {
      id: 'local-' + Date.now(),
      text: input.trim(),
      sender: 'holder',
      timestamp: new Date().toISOString(),
    }]);

    setInput('');
  }

  return (
    <div className="flex flex-col h-screen">
      {/* Header */}
      <header className="px-4 py-3 border-b flex items-center justify-between">
        <h1 className="font-semibold">Token: {code}</h1>
        <button
          onClick={() => window.location.href = \`/t/\${code}/call\`}
          className="p-2 rounded-full hover:bg-gray-100"
          title="Start call"
        >
          📞
        </button>
      </header>

      {/* Messages */}
      <div className="flex-1 overflow-y-auto p-4 space-y-3">
        {messages.map(msg => (
          <div
            key={msg.id}
            className={\`flex \${msg.sender === 'holder' ? 'justify-end' : 'justify-start'}\`}
          >
            <div className={\`max-w-[75%] px-4 py-2 rounded-2xl \${
              msg.sender === 'holder'
                ? 'bg-blue-600 text-white rounded-br-md'
                : 'bg-gray-100 text-gray-900 rounded-bl-md'
            }\`}>
              <p>{msg.text}</p>
              <span className="text-xs opacity-60 mt-1 block">
                {new Date(msg.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
              </span>
            </div>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      {/* Input */}
      <form onSubmit={sendMessage} className="p-4 border-t flex gap-2">
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="Type a message..."
          className="flex-1 px-4 py-2 border rounded-full focus:outline-none focus:ring-2"
        />
        <button
          type="submit"
          disabled={!input.trim()}
          className="px-4 py-2 bg-blue-600 text-white rounded-full disabled:opacity-50"
        >
          Send
        </button>
      </form>
    </div>
  );
}

Voice and video calls in the browser

// web/src/pages/CallPage.tsx
import { useState, useRef, useEffect } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import { useCall } from '../hooks/useCall';
import { useWebSocket } from '../hooks/useWebSocket';

export function CallPage() {
  const { code } = useParams<{ code: string }>();
  const navigate = useNavigate();
  const localVideoRef = useRef<HTMLVideoElement>(null);
  const remoteVideoRef = useRef<HTMLVideoElement>(null);

  const token = sessionStorage.getItem('holderJwt');
  const wsClient = useWebSocket(import.meta.env.VITE_WS_URL, token);
  const { callState, muted, startCall, endCall, toggleMute } = useCall(wsClient.current);

  const [withVideo, setWithVideo] = useState(true);

  useEffect(() => {
    // Auto-start call when page loads
    if (wsClient.current && localVideoRef.current && remoteVideoRef.current) {
      startCall(localVideoRef.current, remoteVideoRef.current, withVideo)
        .catch(err => {
          alert(err.message);
          navigate(\`/t/\${code}/chat\`);
        });
    }
  }, [wsClient.current]);

  function handleEndCall() {
    endCall();
    navigate(\`/t/\${code}/chat\`);
  }

  return (
    <div className="relative h-screen bg-gray-900">
      {/* Remote video — full screen */}
      <video
        ref={remoteVideoRef}
        className="w-full h-full object-cover"
        playsInline
        autoPlay
      />

      {/* Local video — small overlay */}
      <video
        ref={localVideoRef}
        className="absolute top-4 right-4 w-32 h-24 rounded-lg object-cover border-2 border-white"
        playsInline
        autoPlay
        muted
      />

      {/* Call status */}
      {callState === 'calling' && (
        <div className="absolute inset-0 flex items-center justify-center bg-black/50">
          <p className="text-white text-xl">Connecting...</p>
        </div>
      )}

      {/* Controls */}
      <div className="absolute bottom-8 left-1/2 -translate-x-1/2 flex gap-4">
        <button
          onClick={toggleMute}
          className={\`w-14 h-14 rounded-full flex items-center justify-center \${
            muted ? 'bg-red-500' : 'bg-white/20'
          }\`}
        >
          {muted ? '🔇' : '🔈'}
        </button>
        <button
          onClick={handleEndCall}
          className="w-14 h-14 rounded-full bg-red-600 flex items-center justify-center"
        >
          📞
        </button>
      </div>
    </div>
  );
}
The playsInline attribute

On iOS Safari, video elements without playsInline go fullscreen automatically. Add playsInline to all <video> elements for proper inline playback on mobile browsers. Without it, the call UI layout breaks on iPhones.

Progressive enhancement — app-installed detection

If the issuer (who has the app) opens a redemption link, they should be redirected to the native app instead of the web page:

// web/src/lib/deepLink.ts

/**
 * Try to open the native app via deep link.
 * If it fails (app not installed), stay on the web page.
 */
export function tryDeepLink(code: string): Promise<boolean> {
  return new Promise((resolve) => {
    const appUrl = \`tokn://t/\${code}\`;
    const timeout = setTimeout(() => {
      // If we're still here after 1.5s, app didn't open
      resolve(false);
    }, 1500);

    // Try to open the app
    window.location.href = appUrl;

    // If the page loses focus, the app opened
    window.addEventListener('blur', () => {
      clearTimeout(timeout);
      resolve(true);
    }, { once: true });
  });
}

// Usage in RedeemPage — try app first, fall back to web
async function handleRedeemPageLoad(code: string) {
  // Only try deep link on mobile
  const isMobile = /iPhone|iPad|Android/i.test(navigator.userAgent);

  if (isMobile) {
    const appOpened = await tryDeepLink(code);
    if (appOpened) return; // User is in the app now
  }

  // Continue with web redemption flow
}

Universal Links / App Links (better approach)

Deep links with custom schemes are fragile. The better approach uses Universal Links (iOS) and App Links (Android):

// web/public/.well-known/apple-app-site-association
{
  "applinks": {
    "apps": [],
    "details": [{
      "appID": "TEAM_ID.app.tokn.mobile",
      "paths": ["/t/*"]
    }]
  }
}

// web/public/.well-known/assetlinks.json (Android)
[{
  "relation": ["delegate_permission/common.handle_all_urls"],
  "target": {
    "namespace": "android_app",
    "package_name": "app.tokn.mobile",
    "sha256_cert_fingerprints": ["YOUR_SHA256_FINGERPRINT"]
  }
}]

With these files served from tokn.app/.well-known/, the OS intercepts the URL before the browser loads it. If the app is installed, it opens directly. If not, the browser loads the web page normally — no JavaScript detection needed.

SEO and meta tags

When someone shares a tokn.app/t/CODE link in a chat app, it should show a preview:

<!-- web/index.html — meta tags for link previews -->
<head>

  <title>Token — Secure Communication</title>

  <!-- Open Graph (Facebook, WhatsApp, Telegram previews) -->

  <meta property="og:image" content="https://tokn.app/og-image.png" />
  <meta property="og:url" content="https://tokn.app" />
  <meta property="og:type" content="website" />

  <!-- Twitter Card -->
  <meta name="twitter:card" content="summary" />
  <meta name="twitter:title" content="Token — Secure Communication" />
  <meta name="twitter:description" content="Open this link to connect securely." />
  <meta name="twitter:image" content="https://tokn.app/og-image.png" />

  <!-- Security -->
  <meta name="robots" content="noindex" />
  <!-- Don't index individual token pages — they're private -->
</head>
Privacy consideration

Individual token URLs (/t/MERC-8GH2-LP4X) should have noindex so search engines don't crawl them. The OG tags are generic — they don't reveal the token label or issuer identity. The preview just says "secure communication link".

Most holders will open the link on their phone. The layout must work on small screens:

Connection state indicator

Show holders the WebSocket connection status so they know if messages are flowing:

// web/src/components/ConnectionStatus.tsx
import { useState, useEffect } from 'react';

interface Props {
  wsClient: any;
}

export function ConnectionStatus({ wsClient }: Props) {
  const [connected, setConnected] = useState(true);

  useEffect(() => {
    if (!wsClient) return;
    wsClient.onConnect(() => setConnected(true));
    wsClient.onDisconnect(() => setConnected(false));
  }, [wsClient]);

  if (connected) return null; // Don't show anything when connected

  return (
    <div className="fixed top-0 left-0 right-0 bg-orange-500 text-white text-center py-1 text-sm z-50">
      Reconnecting...
    </div>
  );
}

Putting it all together — the complete holder flow

// web/src/App.tsx — final version with all routes
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import { RedeemPage } from './pages/RedeemPage';
import { ChatPage } from './pages/ChatPage';
import { CallPage } from './pages/CallPage';
import { ExpiredPage } from './pages/ExpiredPage';

export default function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/t/:code" element={<RedeemPage />} />
        <Route path="/t/:code/chat" element={<ChatPage />} />
        <Route path="/t/:code/call" element={<CallPage />} />
        <Route path="/t/:code/expired" element={<ExpiredPage />} />
        <Route path="*" element={
          <div className="h-screen flex items-center justify-center">
            <p className="text-gray-500">This link is not valid.</p>
          </div>
        } />
      </Routes>
    </BrowserRouter>
  );
}
