Typing Indicators & Presence

A6 — Lesson 3 of 3 ~45 min
Why this matters for Token

Typing indicators ("typing...") and presence ("online" / "last seen 5 min ago") turn a static message list into a live conversation. They're the difference between "did they see my message?" and knowing the other person is actively there, composing a reply. These signals cost almost nothing in bandwidth but make the chat feel human.

Typing indicator — client-side debounce

You don't want to send a WebSocket message on every keystroke. The pattern: send typing:start on the first keystroke, then reset a 3-second timer on each subsequent keystroke. When the timer expires (no typing for 3 seconds), send typing:stop.

// app/src/hooks/useTypingIndicator.ts
import { useRef, useCallback } from 'react';
import { send } from '../ws/connection';

export function useTypingIndicator(conversationId: number) {
  const isTyping = useRef(false);
  const timeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const handleTextChange = useCallback((text: string) => {
    if (text.length > 0 && !isTyping.current) {
      // First keystroke — notify start
      isTyping.current = true;
      send('typing:start', { conversationId });
    }

    // Reset the stop timer on every change
    if (timeoutRef.current) clearTimeout(timeoutRef.current);

    if (text.length > 0) {
      timeoutRef.current = setTimeout(() => {
        isTyping.current = false;
        send('typing:stop', { conversationId });
      }, 3000);
    } else {
      // Text cleared (e.g. after send) — stop immediately
      if (isTyping.current) {
        isTyping.current = false;
        send('typing:stop', { conversationId });
      }
    }
  }, [conversationId]);

  const stopTyping = useCallback(() => {
    if (timeoutRef.current) clearTimeout(timeoutRef.current);
    if (isTyping.current) {
      isTyping.current = false;
      send('typing:stop', { conversationId });
    }
  }, [conversationId]);

  return { handleTextChange, stopTyping };
}

Displaying "typing..." from the other user

// app/src/components/TypingIndicator.tsx
import { useEffect, useState } from 'react';
import { View, Text, StyleSheet, Animated } from 'react-native';
import { useWsMessage } from '../ws/useWebSocket';

interface TypingIndicatorProps {
  conversationId: number;
  currentUserId: number;
}

export function TypingIndicator({ conversationId, currentUserId }: TypingIndicatorProps) {
  const [isOtherTyping, setIsOtherTyping] = useState(false);

  useWsMessage('typing:update', (payload) => {
    if (
      payload.conversationId === conversationId &&
      payload.userId !== currentUserId
    ) {
      setIsOtherTyping(payload.isTyping);
    }
  });

  // Safety: auto-clear after 5 seconds (in case stop event is lost)
  useEffect(() => {
    if (!isOtherTyping) return;

    const timer = setTimeout(() => setIsOtherTyping(false), 5000);
    return () => clearTimeout(timer);
  }, [isOtherTyping]);

  if (!isOtherTyping) return null;

  return (
    <View style={styles.container}>
      <BouncingDots />
      <Text style={styles.text}>typing</Text>
    </View>
  );
}

function BouncingDots() {
  // Three dots with staggered animation
  const dot1 = new Animated.Value(0);
  const dot2 = new Animated.Value(0);
  const dot3 = new Animated.Value(0);

  useEffect(() => {
    const animate = (dot: Animated.Value, delay: number) =>
      Animated.loop(
        Animated.sequence([
          Animated.delay(delay),
          Animated.timing(dot, { toValue: -4, duration: 300, useNativeDriver: true }),
          Animated.timing(dot, { toValue: 0, duration: 300, useNativeDriver: true }),
        ])
      );

    const a1 = animate(dot1, 0);
    const a2 = animate(dot2, 150);
    const a3 = animate(dot3, 300);
    a1.start(); a2.start(); a3.start();

    return () => { a1.stop(); a2.stop(); a3.stop(); };
  }, []);

  return (
    <View style={styles.dotsRow}>
      {[dot1, dot2, dot3].map((dot, i) => (
        <Animated.View
          key={i}
          style={[styles.dot, { transform: [{ translateY: dot }] }]}
        />
      ))}
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flexDirection: 'row',
    alignItems: 'center',
    paddingHorizontal: 16,
    paddingVertical: 4,
    gap: 6,
  },
  text: {
    fontSize: 13,
    color: '#94a3b8',
    fontStyle: 'italic',
  },
  dotsRow: {
    flexDirection: 'row',
    gap: 3,
  },
  dot: {
    width: 6,
    height: 6,
    borderRadius: 3,
    backgroundColor: '#94a3b8',
  },
});

Presence — online/offline dots

Tracking presence state

// app/src/hooks/usePresence.ts
import { useState, useEffect } from 'react';
import { useWsMessage } from '../ws/useWebSocket';
import { apiRequest } from '../api/client';

interface PresenceState {
  status: 'online' | 'offline';
  lastSeenAt: string | null;
}

export function usePresence(userId: number): PresenceState {
  const [presence, setPresence] = useState<PresenceState>({
    status: 'offline',
    lastSeenAt: null,
  });

  // Fetch initial presence on mount
  useEffect(() => {
    apiRequest<{ data: PresenceState }>('GET', `/users/${userId}/presence`)
      .then(res => setPresence(res.data))
      .catch(() => {}); // not critical — default to offline
  }, [userId]);

  // Listen for real-time updates
  useWsMessage('presence:update', (payload) => {
    if (payload.userId === userId) {
      setPresence({
        status: payload.status,
        lastSeenAt: payload.status === 'offline' ? new Date().toISOString() : null,
      });
    }
  });

  return presence;
}

Displaying presence

// app/src/components/PresenceDot.tsx
import { View, Text, StyleSheet } from 'react-native';

interface PresenceDotProps {
  status: 'online' | 'offline';
  lastSeenAt: string | null;
  showLabel?: boolean;
}

export function PresenceDot({ status, lastSeenAt, showLabel = false }: PresenceDotProps) {
  return (
    <View style={styles.container}>
      <View style={[styles.dot, status === 'online' ? styles.online : styles.offline]} />
      {showLabel && (
        <Text style={styles.label}>
          {status === 'online' ? 'Online' : formatLastSeen(lastSeenAt)}
        </Text>
      )}
    </View>
  );
}

function formatLastSeen(iso: string | null): string {
  if (!iso) return 'Offline';

  const diff = Date.now() - new Date(iso).getTime();
  const minutes = Math.floor(diff / 60000);

  if (minutes < 1) return 'Last seen just now';
  if (minutes < 60) return `Last seen ${minutes}m ago`;

  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `Last seen ${hours}h ago`;

  return `Last seen ${new Date(iso).toLocaleDateString()}`;
}

  },

    backgroundColor: '#94a3b8',

Client-side safety timeout for typing

The server has a 5-second timeout (B5 lesson 3), but the client should also protect itself. If a typing:update { isTyping: true } arrives and no isTyping: false follows within 5 seconds, auto-clear:

// Already in TypingIndicator above:
useEffect(() => {
  if (!isOtherTyping) return;
  const timer = setTimeout(() => setIsOtherTyping(false), 5000);
  return () => clearTimeout(timer);
}, [isOtherTyping]);

This handles edge cases: server crashes mid-typing event, network drops the stop message, or the typing user's app crashes.

Performance — avoiding unnecessary re-renders

Typing events fire frequently. Each typing:update message triggers a state change. Make sure only the TypingIndicator component re-renders — not the entire message list:

// TypingIndicator is a separate component with its own state.
// The MessageThread doesn't own the typing state — TypingIndicator does.
// This means typing:update events DON'T cause FlatList to re-render.

// Pattern: lift state DOWN, not up.
// Each component subscribes to only the WebSocket events it cares about.

Same for presence: the PresenceDot in the header subscribes independently. A presence change doesn't re-render the message list.

Combining it all in the thread

// app/src/screens/MessageThread.tsx — complete structure
export function MessageThreadScreen({ route, navigation }) {
  const { conversationId, tokenLabel, otherUserId } = route.params;
  const { user } = useAuth();
  const [messages, setMessages] = useState([]);
  const [text, setText] = useState('');
  const { handleTextChange, stopTyping } = useTypingIndicator(conversationId);

  // Set custom header with presence
  useEffect(() => {
    navigation.setOptions({
      headerTitle: () => (
        <MessageThreadHeader otherUserId={otherUserId} tokenLabel={tokenLabel} />
      ),
    });
  }, [otherUserId, tokenLabel]);

  // ... message loading, sending, receiving (from lesson 2) ...

  function onChangeText(newText: string) {
    setText(newText);
    handleTextChange(newText);
  }

  function handleSend() {
    if (!text.trim()) return;
    stopTyping();
    // ... optimistic send logic ...
    setText('');
  }

  return (
    <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
      <FlatList data={messages} /* ... */ />

      <TypingIndicator conversationId={conversationId} currentUserId={user!.id} />

      <View style={styles.inputRow}>
        <TextInput
          value={text}
          onChangeText={onChangeText}
          placeholder="Type a message..."
          multiline
        />
        <TouchableOpacity onPress={handleSend} disabled={!text.trim()}>
          <Text>Send</Text>
        </TouchableOpacity>
      </View>
    </KeyboardAvoidingView>
  );
}
