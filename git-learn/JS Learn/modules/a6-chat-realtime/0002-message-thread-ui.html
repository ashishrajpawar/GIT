Message Thread UI

A6 — Lesson 2 of 3 ~60 min
Why this matters for Token

This is where the token holder and issuer actually communicate. The message thread is the payoff for all the infrastructure — token creation, QR scanning, authentication, WebSocket. If this screen feels laggy or confusing, nothing else matters. Messages must appear instantly (optimistic updates), show their delivery status (✓ ✓✓), and scroll smoothly through potentially hundreds of messages.

Message data model (client side)

// app/src/types/message.ts
export interface Message {
  id: number | string;         // server ID (number) or temp local ID (string)
  conversationId: number;
  senderId: number;
  text: string;
  sentAt: string;              // ISO timestamp
  status: 'sending' | 'sent' | 'delivered' | 'read' | 'failed';
}

export interface Conversation {
  id: number;
  tokenId: number;
  tokenLabel: string;
  otherUserId: number;
  otherUserName: string;
  lastMessage: Message | null;
  unreadCount: number;
}

The MessageThread screen

// app/src/screens/MessageThread.tsx
import { useState, useEffect, useRef, useCallback } from 'react';
import {
  View, Text, FlatList, TextInput, TouchableOpacity,
  KeyboardAvoidingView, Platform, StyleSheet,
} from 'react-native';
import { useAuth } from '../auth/AuthContext';
import { useWsMessage } from '../ws/useWebSocket';
import { send } from '../ws/connection';
import { apiRequest } from '../api/client';
import { MessageBubble } from '../components/MessageBubble';

export function MessageThreadScreen({ route }) {
  const { conversationId, tokenLabel } = route.params;
  const { user } = useAuth();
  const [messages, setMessages] = useState<Message[]>([]);
  const [text, setText] = useState('');
  const [loading, setLoading] = useState(true);
  const flatListRef = useRef<FlatList>(null);

  // Load message history via REST
  useEffect(() => {
    loadHistory();
  }, [conversationId]);

  async function loadHistory() {
    const res = await apiRequest<{ data: Message[] }>(
      'GET', `/conversations/${conversationId}/messages?limit=50`
    );
    setMessages(res.data.reverse()); // API returns newest-first, UI needs oldest-first
    setLoading(false);
  }

  // Listen for incoming messages via WebSocket
  useWsMessage('chat:receive', (payload) => {
    if (payload.conversationId === conversationId) {
      const incoming: Message = {
        id: payload.id,
        conversationId: payload.conversationId,
        senderId: payload.senderId,
        text: payload.text,
        sentAt: payload.sentAt,
        status: 'delivered',
      };
      setMessages(prev => [...prev, incoming]);

      // Send read receipt since we're looking at this conversation
      send('chat:read', { messageId: payload.id });
    }
  });

  // Listen for sent acknowledgments
  useWsMessage('chat:sent', (payload) => {
    setMessages(prev =>
      prev.map(m =>
        m.id === payload.localId
          ? { ...m, id: payload.id, sentAt: payload.sentAt, status: 'sent' }
          : m
      )
    );
  });

  // Listen for delivery receipts
  useWsMessage('chat:delivery-receipt', (payload) => {
    setMessages(prev =>
      prev.map(m =>
        m.id === payload.messageId ? { ...m, status: 'delivered' } : m
      )
    );
  });

  function handleSend() {
    if (!text.trim()) return;

    const localId = `local-${Date.now()}`;
    const optimistic: Message = {
      id: localId,
      conversationId,
      senderId: user!.id,
      text: text.trim(),
      sentAt: new Date().toISOString(),
      status: 'sending',
    };

    // Optimistic: add to UI immediately
    setMessages(prev => [...prev, optimistic]);
    setText('');

    // Send via WebSocket
    send('chat:send', {
      conversationId,
      text: text.trim(),
      localId,
    });
  }

  function renderMessage({ item }: { item: Message }) {
    const isMine = item.senderId === user!.id;
    return <MessageBubble message={item} isMine={isMine} />;
  }

  return (
    <KeyboardAvoidingView
      style={styles.container}
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
      keyboardVerticalOffset={90}
    >
      <FlatList
        ref={flatListRef}
        data={messages}
        keyExtractor={item => item.id.toString()}
        renderItem={renderMessage}
        contentContainerStyle={styles.messageList}
        onContentSizeChange={() => flatListRef.current?.scrollToEnd()}
        onLayout={() => flatListRef.current?.scrollToEnd()}
      />

      <View style={styles.inputRow}>
        <TextInput
          style={styles.input}
          value={text}
          onChangeText={setText}
          placeholder="Type a message..."
          multiline
          maxLength={2000}
        />
        <TouchableOpacity
          style={[styles.sendButton, !text.trim() && styles.sendDisabled]}
          onPress={handleSend}
          disabled={!text.trim()}
        >
          <Text style={styles.sendText}>Send</Text>
        </TouchableOpacity>
      </View>
    </KeyboardAvoidingView>
  );
}

// app/src/components/MessageBubble.tsx
import { View, Text, StyleSheet } from 'react-native';
import { Message } from '../types/message';

interface MessageBubbleProps {
  message: Message;
  isMine: boolean;
}

export function MessageBubble({ message, isMine }: MessageBubbleProps) {
  return (
    <View style={[styles.row, isMine && styles.rowMine]}>
      <View style={[styles.bubble, isMine ? styles.bubbleMine : styles.bubbleTheirs]}>
        <Text style={[styles.text, isMine && styles.textMine]}>
          {message.text}
        </Text>
        <View style={styles.meta}>
          <Text style={[styles.time, isMine && styles.timeMine]}>
            {formatTime(message.sentAt)}
          </Text>
          {isMine && <StatusIcon status={message.status} />}
        </View>
      </View>
    </View>
  );
}

function StatusIcon({ status }: { status: Message['status'] }) {
  const icons: Record<string, string> = {
    sending: '○',     // hollow circle — in flight
    sent: '✓',       // single check — server received
    delivered: '✓✓', // double check — recipient device got it
    read: '✓✓',      // double check (blue in real app)
    failed: '!',     // exclamation — send failed
  };
  const color = status === 'read' ? '#3b82f6' : status === 'failed' ? '#ef4444' : '#94a3b8';

  return <Text style={[styles.status, { color }]}>{icons[status]}</Text>;
}

function formatTime(iso: string): string {
  const date = new Date(iso);
  return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
}

Optimistic updates — the pattern

The user taps Send. The message should appear in the thread immediately — before the server responds. This is an optimistic update:

  1. Create a local message with a temporary ID and status 'sending'
  2. Append it to the messages array (UI updates instantly)
  3. Send via WebSocket with the temporary ID attached
  4. On acknowledgment (chat:sent): replace temp ID with server ID, set status to 'sent'
  5. On failure: set status to 'failed', show retry button
// The local ID links the optimistic message to the server acknowledgment
const localId = `local-${Date.now()}`;

// Send with localId so the server can echo it back
send('chat:send', { conversationId, text, localId });

// Server echoes: { type: 'chat:sent', payload: { localId, id: 456, sentAt: '...' } }
// Client maps localId back to the message and updates it

Handling send failures

// If WebSocket is disconnected when user taps Send:
function handleSend() {
  if (!text.trim()) return;

  const localId = `local-${Date.now()}`;
  const optimistic: Message = { /* ... */ status: 'sending' };
  setMessages(prev => [...prev, optimistic]);
  setText('');

  const wsState = getConnectionState();
  if (wsState !== 'connected') {
    // Immediately mark as failed
    setMessages(prev =>
      prev.map(m => m.id === localId ? { ...m, status: 'failed' } : m)
    );
    return;
  }

  send('chat:send', { conversationId, text: text.trim(), localId });

  // Timeout: if no acknowledgment in 10s, mark as failed
  setTimeout(() => {
    setMessages(prev =>
      prev.map(m =>
        m.id === localId && m.status === 'sending'
          ? { ...m, status: 'failed' }
          : m
      )
    );
  }, 10000);
}

KeyboardAvoidingView — platform differences

When the keyboard opens, the input should stay visible (not hidden behind the keyboard):

<KeyboardAvoidingView
  style={{ flex: 1 }}
  behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
  keyboardVerticalOffset={90}  // adjust for header height
>
  {/* FlatList + input */}
</KeyboardAvoidingView>
  • iOS: behavior="padding" works best — adds padding at the bottom to push content up
  • Android: behavior="height" — resizes the view. Note: if android:windowSoftInputMode="adjustResize" is set in AndroidManifest, you might not even need KeyboardAvoidingView on Android
  • keyboardVerticalOffset: accounts for the navigation header height. Adjust based on your header.

Scrolling to bottom on new messages

// Auto-scroll when new messages arrive (but only if already near bottom)
const isNearBottom = useRef(true);

<FlatList
  onScroll={(event) => {
    const { contentOffset, contentSize, layoutMeasurement } = event.nativeEvent;
    const distanceFromBottom = contentSize.height - contentOffset.y - layoutMeasurement.height;
    isNearBottom.current = distanceFromBottom < 100;
  }}
  onContentSizeChange={() => {
    if (isNearBottom.current) {
      flatListRef.current?.scrollToEnd({ animated: true });
    }
  }}
/>

Don't force-scroll if the user has scrolled up to read history. Only auto-scroll when they're near the bottom.

Loading older messages (infinite scroll upward)

const [loadingOlder, setLoadingOlder] = useState(false);
const [hasOlder, setHasOlder] = useState(true);
const oldestId = useRef<number | null>(null);

async function loadOlderMessages() {
  if (loadingOlder || !hasOlder) return;
  setLoadingOlder(true);

  const cursor = oldestId.current;
  const res = await apiRequest<{ data: Message[]; hasMore: boolean }>(
    'GET', `/conversations/${conversationId}/messages?before=${cursor}&limit=30`
  );

  if (res.data.length > 0) {
    oldestId.current = res.data[res.data.length - 1].id as number;
    setMessages(prev => [...res.data.reverse(), ...prev]);
  }
  setHasOlder(res.hasMore);
  setLoadingOlder(false);
}

// Trigger when scrolling to top
<FlatList
  onStartReached={loadOlderMessages}  // react-native-bidirectional-infinite-scroll
  // OR: detect scroll position near top manually
/>

Note: FlatList doesn't have a native onStartReached. You can detect it via the onScroll event (if contentOffset.y < 50, load more) or use inverted={true} with onEndReached (inverted lists are a common chat pattern).

The Conversation List screen

// app/src/screens/ConversationList.tsx
import { useState, useEffect } from 'react';
import { View, Text, FlatList, TouchableOpacity, StyleSheet } from 'react-native';
import { apiRequest } from '../api/client';
import { useWsMessage } from '../ws/useWebSocket';

export function ConversationListScreen({ navigation }) {
  const [conversations, setConversations] = useState<Conversation[]>([]);

  useEffect(() => {
    apiRequest<{ data: Conversation[] }>('GET', '/conversations')
      .then(res => setConversations(res.data));
  }, []);

  // Update last message in real time
  useWsMessage('chat:receive', (payload) => {
    setConversations(prev =>
      prev.map(c =>
        c.id === payload.conversationId
          ? { ...c, lastMessage: payload, unreadCount: c.unreadCount + 1 }
          : c
      ).sort((a, b) =>
        // Most recent conversation first
        new Date(b.lastMessage?.sentAt || 0).getTime() -
        new Date(a.lastMessage?.sentAt || 0).getTime()
      )
    );
  });

  function renderConversation({ item }: { item: Conversation }) {
    return (
      <TouchableOpacity
        style={styles.convRow}
        onPress={() => navigation.navigate('MessageThread', {
          conversationId: item.id,
          tokenLabel: item.tokenLabel,
        })}
      >
        <View style={styles.convInfo}>
          <Text style={styles.convName}>{item.tokenLabel}</Text>
          <Text style={styles.convPreview} numberOfLines={1}>
            {item.lastMessage?.text || 'No messages yet'}
          </Text>
        </View>
        {item.unreadCount > 0 && (
          <View style={styles.badge}>
            <Text style={styles.badgeText}>{item.unreadCount}</Text>
          </View>
        )}
      </TouchableOpacity>
    );
  }

  return (
    <FlatList
      data={conversations}
      keyExtractor={item => item.id.toString()}
      renderItem={renderConversation}
    />
  );
}
