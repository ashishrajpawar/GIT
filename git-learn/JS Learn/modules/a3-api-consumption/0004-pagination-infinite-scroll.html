Pagination & Infinite Scroll

A3 — Lesson 4 of 4 ~45 min
Why this matters for Token

A user with 200 tokens shouldn't wait for all of them to load before seeing anything. Pagination loads 20 at a time, and infinite scroll fetches the next 20 when they reach the bottom. The same pattern applies to message history — you load the most recent messages first, then older ones on scroll up. This is how every professional chat and list app works.

How cursor pagination works

The API returns a cursor — an opaque string that points to the last item. Pass it back to get the next page:

// First request — no cursor
GET /api/tokens?limit=20
Response: { items: [...20 tokens], cursor: "eyJpZCI6MjB9", hasMore: true }

// Next page — include the cursor
GET /api/tokens?limit=20&cursor=eyJpZCI6MjB9
Response: { items: [...20 more tokens], cursor: "eyJpZCI6NDB9", hasMore: true }

// Last page
GET /api/tokens?limit=20&cursor=eyJpZCI6NDB9
Response: { items: [...8 remaining tokens], cursor: null, hasMore: false }

FlatList with onEndReached

React Native's FlatList has built-in infinite scroll support:

import { FlatList, ActivityIndicator } from 'react-native';

function TokenListScreen() {
  const [tokens, setTokens] = useState<TokenListItem[]>([]);
  const [cursor, setCursor] = useState<string | null>(null);
  const [hasMore, setHasMore] = useState(true);
  const [loading, setLoading] = useState(true);
  const [loadingMore, setLoadingMore] = useState(false);

  // Initial load
  useEffect(() => {
    loadTokens();
  }, []);

  async function loadTokens(nextCursor?: string) {
    if (!nextCursor) setLoading(true);
    else setLoadingMore(true);

    try {
      const res = await api.listTokens(nextCursor);
      const { items, cursor: newCursor, hasMore: more } = res.data;

      setTokens(prev => nextCursor ? [...prev, ...items] : items);
      setCursor(newCursor);
      setHasMore(more);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
      setLoadingMore(false);
    }
  }

  function handleEndReached() {
    if (!loadingMore && hasMore && cursor) {
      loadTokens(cursor);
    }
  }

  return (
    <FlatList
      data={tokens}
      keyExtractor={item => item.code}
      renderItem={({ item }) => <TokenCard token={item} />}
      onEndReached={handleEndReached}
      onEndReachedThreshold={0.5}
      ListFooterComponent={
        loadingMore ? <ActivityIndicator style={{ padding: 16 }} /> : null
      }
    />
  );
}

Key props explained

  • onEndReached — fires when the user scrolls near the bottom
  • onEndReachedThreshold — how close to the bottom (0.5 = halfway through the last screen-height) before triggering
  • ListFooterComponent — shows a spinner below the list while loading more

Pull-to-refresh

<FlatList
  data={tokens}
  onRefresh={() => loadTokens()}    // reload from first page
  refreshing={loading}              // show pull-to-refresh spinner
  onEndReached={handleEndReached}
  ...
/>

Pull-to-refresh resets the list (loads page 1 fresh). Infinite scroll appends new pages to the existing list.

Preventing duplicate fetches

Without guards, onEndReached can fire multiple times rapidly:

function handleEndReached() {
  // Guard: don't fetch if already loading or no more pages
  if (loadingMore || !hasMore || !cursor) return;
  loadTokens(cursor);
}

Message history — reverse pagination

Chat messages load newest-first (what you see on screen) and paginate upward (older messages on scroll up):

function MessageThread({ conversationId }: Props) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [cursor, setCursor] = useState<string | null>(null);
  const [hasMore, setHasMore] = useState(true);
  const [loadingMore, setLoadingMore] = useState(false);

  useEffect(() => {
    loadMessages();
  }, [conversationId]);

  async function loadMessages(nextCursor?: string) {
    setLoadingMore(true);
    const res = await api.getMessages(conversationId, nextCursor);
    const { items, cursor: newCursor, hasMore: more } = res.data;

    // Prepend older messages (they go above current ones)
    setMessages(prev => nextCursor ? [...items, ...prev] : items);
    setCursor(newCursor);
    setHasMore(more);
    setLoadingMore(false);
  }

  return (
    <FlatList
      data={messages}
      inverted            // newest at bottom (chat convention)
      keyExtractor={item => String(item.id)}
      renderItem={({ item }) => <MessageBubble message={item} />}
      onEndReached={() => {
        if (!loadingMore && hasMore && cursor) loadMessages(cursor);
      }}
      onEndReachedThreshold={0.3}
      ListFooterComponent={
        loadingMore ? <ActivityIndicator /> : null
      }
    />
  );
}

inverted flips the FlatList — newest messages at the bottom, scrolling up loads older ones. onEndReached fires when reaching the "top" (which is the visual bottom in an inverted list).

Custom hook: usePaginatedApi

// app/src/hooks/usePaginatedApi.ts
export function usePaginatedApi<T>(
  fetcher: (cursor?: string) => Promise<{ items: T[]; cursor: string | null; hasMore: boolean }>
) {
  const [items, setItems] = useState<T[]>([]);
  const [cursor, setCursor] = useState<string | null>(null);
  const [hasMore, setHasMore] = useState(true);
  const [loading, setLoading] = useState(true);
  const [loadingMore, setLoadingMore] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async (nextCursor?: string) => {
    if (nextCursor) setLoadingMore(true);
    else setLoading(true);

    try {
      const result = await fetcher(nextCursor);
      setItems(prev => nextCursor ? [...prev, ...result.items] : result.items);
      setCursor(result.cursor);
      setHasMore(result.hasMore);
      setError(null);
    } catch (err: any) {
      setError(err.message);
    } finally {
      setLoading(false);
      setLoadingMore(false);
    }
  }, []);

  useEffect(() => { load(); }, []);

  const loadMore = () => {
    if (!loadingMore && hasMore && cursor) load(cursor);
  };

  const refresh = () => load();

  return { items, loading, loadingMore, error, hasMore, loadMore, refresh };
}

// Usage:
const { items: tokens, loading, loadingMore, loadMore, refresh } = usePaginatedApi(
  (cursor) => api.listTokens(cursor).then(r => r.data)
);
