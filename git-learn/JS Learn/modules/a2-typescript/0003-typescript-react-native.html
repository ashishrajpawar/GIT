TypeScript with React Native

A2 — Lesson 3 of 3 ~50 min
Why this matters for Token

Every component in Token's app/ folder is a .tsx file. Typing your props, navigation params, and event handlers means the editor autocompletes everything and catches errors as you type — not when you test on the phone. This lesson connects the TypeScript you just learned to the React Native you already know.

Typing component props

In Module 2, you wrote components that receive props. Now you type them:

import { View, Text, Pressable } from 'react-native';

interface TokenCardProps {
  code: string;
  issuedTo: string;
  status: 'active' | 'paused' | 'revoked';
  onPress: () => void;
}

function TokenCard({ code, issuedTo, status, onPress }: TokenCardProps) {
  return (
    <Pressable onPress={onPress}>
      <View>
        <Text>{code}</Text>
        <Text>{issuedTo}</Text>
        <Text>{status}</Text>
      </View>
    </Pressable>
  );
}

If a parent component forgets to pass onPress or passes the wrong type, the editor shows an error before you even save the file.

Optional props and defaults

interface TokenBadgeProps {
  status: 'active' | 'paused' | 'revoked';
  size?: 'small' | 'large';  // optional — defaults to 'small'
}

function TokenBadge({ status, size = 'small' }: TokenBadgeProps) {
  const fontSize = size === 'large' ? 16 : 12;
  return <Text style={{ fontSize }}>{status}</Text>;
}

Typing useState

TypeScript usually infers the state type from the initial value. But when the initial value doesn't show the full type, annotate it:

import { useState } from 'react';
import { Token } from '@token/shared/types';

function TokenListScreen() {
  // Inferred as string — fine
  const [searchQuery, setSearchQuery] = useState('');

  // Inferred as number — fine
  const [selectedIndex, setSelectedIndex] = useState(0);

  // Initial value is [] — TS can't infer what's in the array.
  // Annotate explicitly:
  const [tokens, setTokens] = useState<Token[]>([]);

  // Nullable state — starts null, later holds a Token:
  const [selectedToken, setSelectedToken] = useState<Token | null>(null);

  return null; // ... UI
}

Typing navigation params

React Navigation uses a type parameter to know what params each screen accepts:

import { NativeStackScreenProps } from '@react-navigation/native-stack';

// Define all screens and their params
type RootStackParamList = {
  Login: undefined;              // no params
  Register: undefined;
  Main: undefined;
  TokenDetail: { tokenCode: string };  // requires tokenCode
  CreateToken: { prefill?: string };   // optional prefill
};

// Type a screen component:
type TokenDetailProps = NativeStackScreenProps<RootStackParamList, 'TokenDetail'>;

function TokenDetailScreen({ route, navigation }: TokenDetailProps) {
  const { tokenCode } = route.params;  // TS knows this is string

  // navigation.navigate is typed — only valid screen names allowed:
  navigation.navigate('Main');  // OK
  // navigation.navigate('Nonexistent');  // Error!

  return null;
}

Typing event handlers

import { TextInput, NativeSyntheticEvent, TextInputChangeEventData } from 'react-native';

function SearchBar() {
  const [query, setQuery] = useState('');

  // onChangeText gives you the string directly — simplest
  const handleChange = (text: string) => {
    setQuery(text);
  };

  // For more complex events (onSubmitEditing, onFocus):
  const handleSubmit = (e: NativeSyntheticEvent<TextInputChangeEventData>) => {
    console.log('Submitted:', e.nativeEvent.text);
  };

  return (
    <TextInput
      value={query}
      onChangeText={handleChange}
      onSubmitEditing={handleSubmit}
    />
  );
}

Typing FlatList renderItem

import { FlatList, ListRenderItem } from 'react-native';

interface Token {
  code: string;
  issuedTo: string;
  status: 'active' | 'paused' | 'revoked';
}

function TokenList({ tokens }: { tokens: Token[] }) {
  const renderItem: ListRenderItem<Token> = ({ item }) => (
    <TokenCard
      code={item.code}        // TS knows item is Token
      issuedTo={item.issuedTo}
      status={item.status}
      onPress={() => {}}
    />
  );

  return (
    <FlatList
      data={tokens}
      renderItem={renderItem}
      keyExtractor={(item) => item.code}
    />
  );
}

Typing styles

import { StyleSheet, ViewStyle, TextStyle, ImageStyle } from 'react-native';

// StyleSheet.create infers types automatically:
const styles = StyleSheet.create({
  container: {
    flex: 1,
    padding: 16,
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold',
  },
});

// For dynamic styles, type them explicitly:
function getStatusStyle(status: 'active' | 'paused' | 'revoked'): ViewStyle {
  const colors = { active: '#dcfce7', paused: '#fef3c7', revoked: '#fee2e2' };
  return {
    backgroundColor: colors[status],
    padding: 8,
    borderRadius: 6,
  };
}

Common patterns in Token

Custom hooks with return types

function useTokens() {
  const [tokens, setTokens] = useState<Token[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // The return type is inferred, but you could annotate it:
  return { tokens, loading, error, refetch: () => {} };
}

// Usage — TS knows exactly what's returned:
const { tokens, loading } = useTokens();

Children prop

import { ReactNode } from 'react';

interface ScreenWrapperProps {
  title: string;
  children: ReactNode;
}

function ScreenWrapper({ title, children }: ScreenWrapperProps) {
  return (
    <View>
      <Text>{title}</Text>
      {children}
    </View>
  );
}
